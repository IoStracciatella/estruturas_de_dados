As expressões *p++ e (*p)++ são diferentes porque, na primeira, incrementamos o ponteiro, fazendo-o apontar para a próxima posição de memória, e, na segunda, incrementamos o valor no bloco de memória. No caso do código que está no exercício, a saída é abc a = abc a = dbc, afinal, no segundo caso mudamos o valor da variável no bloco de memória

#include <stdio.h>

int main() {
    int i, soma = 0, elementos = 0, numeros[15];
    float media;
    
    for (i = 0; i < 15; i++) {
        scanf("%d", &numeros[i]);
    }
    
    for (i = 0; i < 15; i++) {
        if (i % 2 == 0) {
            printf("%d", numeros[i]);
            soma += numeros[i];
            elementos++;
        }
    }
    
    media = (float) soma / (float) elementos;
    
    printf("Média: %f", media);

    return 0;
}

#include <stdio.h>

int main() {
    int i, maior_alt = 0, menor_alt, soma, qtde_n_fum = 0, qtde_fum = 0, media, altura[10], fumante[10];
    
    for (i = 0; i < 10; i++) {
        printf("Insira a altura (em cm) e se eh fumante (0 = nao 1 = sim):\n");
        scanf("%d %d", &altura[i], &fumante[i]);
        
        if (fumante[i] < 0 || fumante[i] > 1 || altura[i] <= 0) {
            printf("Insira um valor valido");
            return 1;
        }
        
        if (maior_alt < altura[i]) {
            maior_alt = altura[i];
        }
        
        if (fumante[i] == 0) {
            soma += altura[i];
            qtde_n_fum++;
        }
    }
    
    menor_alt = altura[0];
    
    for (i = 0; i < 10; i++) {
        if (menor_alt > altura[i]) {
            menor_alt = altura[i];
        }
    }
    
    media = soma/qtde_n_fum;
    
    qtde_fum = 10 - qtde_n_fum;
    
    printf("Maior altura: %d\nMenor altura: %d\nMedia nao fumantes: %d\nQuantidade de fumantes: %d", maior_alt, menor_alt, media, qtde_fum);

    return 0;
}
